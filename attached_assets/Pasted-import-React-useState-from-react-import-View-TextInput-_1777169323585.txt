import React, { useState } from 'react';
import { View, TextInput, TouchableOpacity, KeyboardAvoidingView, Platform } from 'react-native';
import { Ionicons } from '@expo/vector-icons'; // For the 'send' and 'attach' icons

export const StarLightMessenger = ({ deviceId }) => {
  const [message, setMessage] = useState('');

  const handleSend = async () => {
    if (message.trim().length === 0) return;

    // 1. Encrypt the text before it leaves the device
    const encryptedPacket = await encryptStarPacket(JSON.stringify({
      text: message,
      sender: "Dennis_Admin",
      timestamp: Date.now()
    }));

    // 2. Beam it via Bluetooth Mesh
    await sendStarLightPacket(deviceId, 'TEXT', encryptedPacket);

    // 3. Clear the box
    setMessage('');
    console.log("Message scrambled and beamed.");
  };

  return (
    <KeyboardAvoidingView 
      behavior={Platform.OS === 'ios' ? 'padding' : 'height'}
      style={{ position: 'absolute', bottom: 20, width: '100%', paddingHorizontal: 15 }}
    >
      <View style={{ 
        flexDirection: 'row', 
        backgroundColor: '#1A1A1A', // Matches your dark 'Stealth' theme
        borderRadius: 25, 
        paddingHorizontal: 15, 
        paddingVertical: 10,
        alignItems: 'center',
        borderWidth: 1,
        borderColor: '#00D4FF' // Neon blue 'StarLight' accent
      }}>
        
        {/* Attachment Button */}
        <TouchableOpacity onPress={() => sendFileAttachment(deviceId)}>
          <Ionicons name="add-circle-outline" size={28} color="#00D4FF" />
        </TouchableOpacity>

        {/* The Typing Area */}
        <TextInput
          style={{ flex: 1, color: '#FFF', fontSize: 16, marginHorizontal: 10 }}
          placeholder="Type secure message..."
          placeholderTextColor="#666"
          value={message}
          onChangeText={setMessage}
          multiline
        />

        {/* Send Button */}
        <TouchableOpacity onPress={handleSend}>
          <Ionicons name="send" size={24} color="#00D4FF" />
        </TouchableOpacity>
      </View>
    </KeyboardAvoidingView>
  );
};
