  View, TextInput, TouchableOpacity, KeyboardAvoidingView,

      {/* 1. THE STICKY TOOLBAR (Highlighted on top of keyboard) */}
      <InputAccessoryView nativeID={inputID}>
        <View style={styles.toolbar}>
          
          {/* AI ASSISTANT ICON */}
          <TouchableOpacity onPress={() => console.log("AI Optimized")}>

            <Ionicons name="camera-outline" size={26} color="#00D4FF" />
          </TouchableOpacity>

          {/* ATTACHMENTS (DOCS/FILES) */}
          <TouchableOpacity onPress={() => console.log("File Picker")}>
            <Ionicons name="attach-outline" size={28} color="#00D4FF" />
          </TouchableOpacity>

          {/* EMOJI SELECTOR */}

            <MaterialCommunityIcons name="reply-outline" size={26} color="#00D4FF" />
          </TouchableOpacity>

        </View>
      </InputAccessoryView>

      {/* 2. THE MAIN INPUT FIELD */}
